from openpyxl.utils import get_column_letter

output = root / "TaskBoard-manual-test-cases.xlsx"

if not output.exists():
    wb = Workbook()
    ws = wb.active
    ws.title = "Test Cases"
    headers = ["ID","Priority","Type","Scenario","Preconditions","Steps","Test data","Expected result",
               "Playwright or script result","Observed result","Evidence","Manual status","Manual actual result","Manual tester","Defect"]
    ws.append(headers)
    for row in cases:
        ws.append(list(row) + ["Not run", "", "", "Not run", "", "", ""])
    ws.freeze_panes = "D2"
    ws.auto_filter.ref = ws.dimensions
    for col, width in enumerate([14,10,16,36,36,52,32,55,22,55,30,18,40,22,20], 1):
        ws.column_dimensions[get_column_letter(col)].width = width
    for cell in ws[1]:
        cell.fill = PatternFill("solid", fgColor="17365D")
        cell.font = Font(color="FFFFFF", bold=True)
        cell.alignment = Alignment(wrap_text=True, vertical="top")
    for row in ws.iter_rows(min_row=2):
        for cell in row:
            cell.alignment = Alignment(wrap_text=True, vertical="top")
    validation = DataValidation(type="list", formula1='"Not run,Pass,Fail,Blocked"', allow_blank=False)
    ws.add_data_validation(validation)
    validation.add(f"L2:L{ws.max_row}")
    info = wb.create_sheet("Instructions")
    for line in [
        "TaskBoard manual test cases",
        "Use only the isolated QA instance supplied in the handoff; avoid production data.",
        "Playwright or script results are separate from human manual results. Fill manual status, actual result and tester only after your own execution.",
        "P0/P1 are test priorities, not a severity judgment. Do not copy tokens, credentials or password hashes into the workbook.",
        "Performance is a bounded local baseline, not an SLA or capacity certification.",
        "Known defects are documented separately; an observed failure must not be changed to Pass without evidence.",
    ]:
        info.append([line])
    info.column_dimensions["A"].width = 110
    for row in info.iter_rows():
        row[0].alignment = Alignment(wrap_text=True, vertical="top")
    wb.save(output)
else:
    wb = load_workbook(output)
    ws = wb["Test Cases"]
    existing = {row[0].value for row in ws.iter_rows(min_row=2)}
    for row in cases:
        if row[0] not in existing:
            ws.append(list(row) + ["Not run", "", "", "Not run", "", "", ""])
            for cell in ws[ws.max_row]:
                cell.alignment = Alignment(wrap_text=True, vertical="top")
        else:
            for existing_row in ws.iter_rows(min_row=2):
                if existing_row[0].value == row[0]:
                    existing_row[7].value = row[7]
                    break
    ws.auto_filter.ref = ws.dimensions
    for validation in ws.data_validations.dataValidation:
        validation.sqref = f"L2:L{ws.max_row}"

if len(sys.argv) > 1:
    results = json.loads(Path(sys.argv[1]).read_text())
    for row in ws.iter_rows(min_row=2):
        result = results.get(row[0].value)
        if result:
            row[8].value = result["status"]
            row[9].value = result["actual"]
            row[10].value = result.get("evidence", result.get("executor", ""))
            defects = {"SEC-02": "BUG-01", "SEC-01": "BUG-02", "API-07": "BUG-03",
                       "API-08": "BUG-03", "API-11": "BUG-04"}
            row[14].value = result.get("defect", defects.get(row[0].value, ""))
wb.save(output)
print(f"{output}: {ws.max_row - 1} cases")

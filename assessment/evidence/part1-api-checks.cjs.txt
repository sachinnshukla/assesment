const fs = require("node:fs");
const { performance } = require("node:perf_hooks");

const base = "http://localhost:3101";
const results = {};
const record = (id, passed, actual) => {
  results[id] = { status: passed ? "Pass" : "Fail", actual, executor: "API script" };
  console.log(`${id}: ${results[id].status} - ${actual}`);
};

async function request(path, { token, method = "GET", body } = {}) {
  const headers = {};
  if (token) headers.Authorization = `Bearer ${token}`;
  if (body !== undefined) headers["Content-Type"] = "application/json";
  const response = await fetch(`${base}${path}`, {
    method,
    headers,
    ...(body !== undefined ? { body: JSON.stringify(body) } : {}),
  });
  const text = await response.text();
  return { status: response.status, data: text ? JSON.parse(text) : null };
}

async function login(email) {
  const response = await request("/api/auth/login", {
    method: "POST",
    body: { email, password: "password123" },
  });
  if (response.status !== 200 || !response.data?.token) throw new Error(`Login failed: ${response.status}`);
  return response.data.token;
}

async function main() {
  const [admin, viewer, outsider] = await Promise.all([
    login("meera@taskboard.dev"), login("dev@example.com"), login("lina@example.com"),
  ]);
  record("API-01", true, "Valid seeded admin login returned HTTP 200 and a token.");
  const bad = await request("/api/auth/login", {
    method: "POST", body: { email: "meera@taskboard.dev", password: "incorrect-password" },
  });
  record("API-02", bad.status === 401, `Invalid password returned HTTP ${bad.status}.`);
  const noAuth = await request("/api/projects");
  record("API-03", noAuth.status === 401, `Unauthenticated project request returned HTTP ${noAuth.status}.`);
  const projects = await request("/api/projects", { token: viewer });
  const list = projects.data?.projects;
  record("API-04", projects.status === 200 && Array.isArray(list) && list.length === 1 && list[0].name === "Q3 Launch",
    `Viewer project list returned HTTP ${projects.status}, ${Array.isArray(list) ? list.length : "unknown"} project(s).`);
  const adminProjects = await request("/api/projects", { token: admin });
  const launch = adminProjects.data.projects.find((p) => p.name === "Q3 Launch");
  const privateProject = adminProjects.data.projects.find((p) => p.name === "Internal Tools Cleanup");
  if (!launch || !privateProject) throw new Error("Missing required isolated seed projects");
  const forbidden = await request(`/api/projects/${privateProject.id}`, { token: viewer });
  record("API-05", forbidden.status === 403, `Non-member project detail returned HTTP ${forbidden.status}.`);
  const create = await request(`/api/projects/${launch.id}/tasks`, {
    token: viewer, method: "POST", body: { title: "QA viewer create attempt" },
  });
  record("API-06", create.status === 403, `Viewer task creation returned HTTP ${create.status}.`);
  const tasks = await request(`/api/projects/${launch.id}/tasks`, { token: admin });
  if (!tasks.data?.tasks?.length) throw new Error("Missing isolated seed tasks");
  const task = tasks.data.tasks[0];
  const update = await request(`/api/tasks/${task.id}`, {
    token: viewer, method: "PATCH", body: { title: "QA viewer update attempt" },
  });
  record("API-07", update.status === 403, `Viewer task update returned HTTP ${update.status}; expected 403.`);
  const outsiderUpdate = await request(`/api/tasks/${task.id}`, {
    token: outsider, method: "PATCH", body: { title: "QA nonmember update attempt" },
  });
  record("API-08", outsiderUpdate.status === 403, `Non-member task update returned HTTP ${outsiderUpdate.status}; expected 403.`);
  const detail = await request(`/api/projects/${launch.id}`, { token: viewer });
  function containsPasswordHash(value) {
    if (!value || typeof value !== "object") return false;
    return Object.entries(value).some(([k, v]) => k === "passwordHash" || containsPasswordHash(v));
  }
  const exposes = containsPasswordHash(detail.data);
  record("SEC-01", detail.status === 200 && !exposes,
    `Project detail returned HTTP ${detail.status}; passwordHash field ${exposes ? "present" : "absent"} (values omitted).`);
  const injected = await request(`/api/projects/${launch.id}/tasks?q=${encodeURIComponent("x%') OR '1'='1' -- ")}`, { token: viewer });
  const crossProject = Array.isArray(injected.data?.tasks)
    ? injected.data.tasks.filter((t) => t.project_id !== launch.id).length : 0;
  record("SEC-02", injected.status === 200 && crossProject === 0,
    `Search returned HTTP ${injected.status}, ${crossProject} task(s) from other projects; literal search must not bypass project filter.`);
  const invalid = await request("/api/auth/register", {
    method: "POST", body: { email: "not-an-email", name: "QA", password: "short" },
  });
  record("API-09", invalid.status === 400, `Invalid registration returned HTTP ${invalid.status}.`);
  const onboarding = adminProjects.data.projects.find((p) => p.name === "Customer Onboarding Revamp");
  if (!onboarding) throw new Error("Missing isolated onboarding project");
  const onboardingDetail = await request(`/api/projects/${onboarding.id}`, { token: admin });
  const lina = onboardingDetail.data?.project?.memberships?.find((m) => m.user?.email === "lina@example.com")?.user;
  if (!lina) throw new Error("Missing isolated non-member user");
  const assign = await request(`/api/projects/${launch.id}/tasks`, {
    token: admin, method: "POST", body: { title: "QA nonmember assignee test", assigneeId: lina.id },
  });
  results["API-10"] = {
    status: "Blocked",
    actual: `Task creation with a non-member assignee returned HTTP ${assign.status}; whether this is prohibited requires a product decision.`,
    executor: "API script",
  };
  console.log(`API-10: Blocked - ${results["API-10"].actual}`);
  const timings = [];
  let errors = 0;
  for (let batch = 0; batch < 10; batch++) {
    await Promise.all(Array.from({ length: 2 }, async () => {
      const start = performance.now();
      try {
        const r = await request("/api/projects", { token: admin });
        if (r.status !== 200) errors++;
      } catch {
        errors++;
      } finally {
        timings.push(performance.now() - start);
      }
    }));
  }
  timings.sort((a, b) => a - b);
  record("PERF-01", errors === 0, `Read-only local smoke: 20 requests, concurrency 2, ${errors} errors, p50 ${timings[9].toFixed(1)} ms, p95 ${timings[18].toFixed(1)} ms; no latency SLA assumed.`);
  fs.writeFileSync(process.argv[2], JSON.stringify(results, null, 2));
}

main().catch((error) => { console.error(error.message); process.exitCode = 1; });
